파이프라인을 실제 저장소에 붙이기 전에 확인할 것들입니다. 하나라도 빠지면 나중에 조용히 어긋납니다.

확인왜빠졌을 때 증상
절 제목이 표준인가 (## 문제 본문 / ## ✅ 정답 / ## 📝 해설)qbank 파서가 이 이름으로 자릅니다인쇄에 본문이 안 나옴

파일명 규칙(과목_연도_번호.md)정렬·중복 검사·매핑에 쓰입니다목록 정렬이 뒤죽박죽
외부 API로 보내기 전에 한 번 더

문항 원문·모의고사 지문은 내부 자료입니다. 어떤 자료는 출제 기관·출판사의 저작물이고, 어떤 자료는 “외부 공유 금지” 약속이 붙어 있습니다. 노트북에서 API를 호출하면 그 텍스트가 외부로 전송됩니다. 무엇을 보내도 되는지 먼저 정하고, 민감한 자료는 전송 전에 마스킹하거나 로컬 모델로 처리하세요. 기술적으로 쉬운 일과 해도 되는 일은 다릅니다. 이 노트의 코드는 “돌아가는 방법”이지 “보내도 된다”는 허가가 아닙니다.

A.5 자주 쓰는 스니펫

본문에서 만든 함수들을 한 곳에 모았습니다. 새 노트북을 열 때 이 블록부터 붙이면 됩니다.

# ===== 0. 공통 =====
import os, re, json
from pathlib import Path
import anthropic

client = anthropic.Anthropic(api_key=os.environ["ANTHROPIC_API_KEY"])
MODEL = os.environ.get("CLAUDE_MODEL", "claude-sonnet-5-5")

def get_text(resp) -> str:
    return "".join(b.text for b in resp.content if b.type == "text")

def extract_tag(text: str, tag: str):
    m = re.search(rf"<{tag}>(.*?)</{tag}>", text, re.S)
    return m.group(1).strip() if m else None


# ===== 1. 프롬프트 =====
def call_json(user_prompt: str, *, retries: int = 1) -> dict:
    messages = [{"role": "user", "content": user_prompt},
                {"role": "assistant", "content": "<json>{"}]
    resp = client.messages.create(model=MODEL, max_tokens=2000, temperature=0,
                                  messages=messages, stop_sequences=["</json>"])
    raw = "{" + get_text(resp)
    try:
        return json.loads(raw)
    except json.JSONDecodeError:
        if retries <= 0:
            raise
        fix = user_prompt + "\n\n<correction>유효한 JSON만 <json>{ … }</json> 로 출력하시오.</correction>"
        return call_json(fix, retries=retries - 1)


# ===== 2. 문서 =====
def load_item(md_path: Path) -> dict:
    text = md_path.read_text(encoding="utf-8")
    meta, body = {}, text
    if text.startswith("---"):
        parts = text.split("---", 2)
        if len(parts) == 3:
            _, fm, body = parts
            for line in fm.splitlines():
                if line and not line[0].isspace() and ":" in line:
                    k, v = line.split(":", 1)
                    meta[k.strip()] = v.strip().strip('"').strip("'")
    return {"meta": meta, "body": body.strip(), "path": md_path.name}

def section(body: str, *names: str) -> str:
    for name in names:
        m = re.search(rf"^##\s+.*{re.escape(name)}.*$([\s\S]*?)(?=^##\s|\Z)", body, re.M)
        if m:
            return m.group(1).strip()
    return ""


# ===== 3. 검색 =====
def embed_texts(texts, batch_size=16):
    return model.encode(texts, batch_size=batch_size, normalize_embeddings=True,
                        show_progress_bar=True, convert_to_numpy=True).astype("float32")

def rrf_merge(result_lists, k=60, top_n=8):
    fused = {}
    for results in result_lists:
        for rank, (idx, score, chunk) in enumerate(results, start=1):
            slot = fused.setdefault(idx, {"idx": idx, "chunk": chunk, "rrf": 0.0})
            slot["rrf"] += 1.0 / (k + rank)
    return [(d["idx"], d["rrf"], d["chunk"]) for d in
            sorted(fused.values(), key=lambda d: -d["rrf"])[:top_n]]


# ===== 4. 채점 =====
def parse_judge(raw: str) -> dict:
    met = re.findall(r'<criterion id="(\w+)">\s*<met>(true|false|unknown)</met>', raw)
    if not met:
        return {"ok": False}
    return {"ok": True, "met": dict(met),
            "rationale": (re.search(r"<rationale>(.*?)</rationale>", raw, re.S) or [None, ""])[1].strip()}

def score_of(parsed: dict, rubric: dict) -> dict:
    earned = sum(c["points"] for c in rubric["criteria"]
                 if parsed["met"].get(c["id"]) == "true")
    for p in rubric.get("penalties", []):
        if parsed["met"].get(p["id"]) == "true":
            earned += p["points"]
    return {"score": max(rubric.get("floor", 0), min(rubric["total"], earned)), "max": rubric["total"]}

💡 마지막으로, 이 노트를 쓰는 태도
